# Backend development + test dependencies.
#
# 2026-05-22 (audit P2) — the end-to-end validation found that the
# backend pytest suite couldn't run because pytest itself wasn't
# pinned anywhere. requirements.txt is the runtime-only pin
# (FastAPI + DuckDB + drivers etc.); this file adds the test runner
# and developer tooling on top.
#
# Install:
#   cd backend
#   python -m venv .venv
#   .\.venv\Scripts\python -m pip install -r requirements.txt -r requirements-dev.txt
#
# Or use the helper script:
#   .\scripts\setup-backend-tests.ps1

# Test runner + async support
pytest>=8.0
pytest-asyncio>=0.23
pytest-cov>=4.1
pytest-mock>=3.12
# Parallel test execution (2026-06-08, reviewer rec #1): the default
# suite (~3000 tests) runs ~12 min serially — past a typical 10-min CI
# window. conftest.py already gives every test a unique data dir, so the
# suite is xdist-safe; run it parallel with `pytest -n auto`.
pytest-xdist>=3.5

# Auto-retry transient failures in the parallel fast gate (2026-06-19): xdist
# worker crashes (C-extension memory pressure) and import-order registration
# races pass on a clean re-run. Enabled via a pytest_configure hook in
# tests/conftest.py (NOT pytest.ini addopts — the security-scan job runs
# `pytest --noconftest` without this plugin and would choke on a global
# --reruns). A transient flake retries up to twice; a test that fails every
# attempt still fails. Required wherever the conftest-loaded suite runs.
pytest-rerunfailures>=12.0

# HTTP client used by integration tests against a running app
# (httpx is already in requirements.txt for runtime, but pin a
# minimum here too so dev installs work standalone)
httpx>=0.27.0

# Coverage + report formats
coverage>=7.4

# Static analysis (optional but useful in CI)
ruff>=0.5.0

# Type stubs for common deps
types-requests>=2.33.0.20260712
